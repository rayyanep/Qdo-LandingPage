# 2048 — Mobile (iOS + Android)

A polished, animated 2048 clone built with **React Native + Expo**.
Single codebase, swipe gestures, sliding-tile animations, persistent best
score, undo, and "continue past 2048" mode.

---

## 1. Tech Stack

| Layer | Choice | Why |
|-------|--------|-----|
| Framework | React Native (Expo SDK 51+) | One codebase → iOS + Android |
| Language | TypeScript | Type-safe engine + tile identity |
| Gestures | `react-native-gesture-handler` | Swipe detection |
| Animation | `react-native-reanimated` v3 | 60 fps slides on the UI thread |
| Storage | `@react-native-async-storage/async-storage` | Best score + saved game |
| Haptics | `expo-haptics` | Subtle merge feedback |
| Fonts | `expo-font` (Inter / system fallback) | Clean numerals |

## 5. Game Engine (Pure)

Every tile keeps a stable **id** across moves so the UI can animate the
*same* tile sliding from its old position to its new one. Merges record
the two source tiles so we can fade them out and pop the result.

**`src/game/types.ts`**

```ts
export type Direction = 'up' | 'down' | 'left' | 'right';

export type Tile = {
  id: number;        // stable across moves
  value: number;
  row: number;       // current logical position
  col: number;
  prevRow?: number;  // for animation (where it came from)
  prevCol?: number;
  isNew?: boolean;
  mergedFromIds?: [number, number]; // for merge pop
};

export type GameState = {
  tiles: Tile[];     // canonical list (length 1..size*size)
  size: number;      // 4 by default
  score: number;
  best: number;
  status: 'playing' | 'won' | 'lost' | 'continue';
  hasReached2048: boolean;
  history: Omit<GameState, 'history' | 'best'>[]; // for undo (capped)
};
```

**`src/game/engine.ts`**

```ts
import { Direction, GameState, Tile } from './types';

const HISTORY_CAP = 1;          // 1-step undo (raise for more)
const WIN_VALUE = 2048;
let nextId = 1;

const newId = () => nextId++;

function emptyGrid(size: number): (Tile | null)[][] {
  return Array.from({ length: size }, () => Array<Tile | null>(size).fill(null));
}

function gridFromTiles(size: number, tiles: Tile[]): (Tile | null)[][] {
  const g = emptyGrid(size);
  for (const t of tiles) g[t.row][t.col] = t;
  return g;
}

function emptyCells(grid: (Tile | null)[][]): [number, number][] {
  const out: [number, number][] = [];
  for (let r = 0; r < grid.length; r++)
    for (let c = 0; c < grid.length; c++)
      if (!grid[r][c]) out.push([r, c]);
  return out;
}

export function spawnTile(state: GameState): GameState {
  const grid = gridFromTiles(state.size, state.tiles);
  const empties = emptyCells(grid);
  if (empties.length === 0) return state;
  const [r, c] = empties[Math.floor(Math.random() * empties.length)];
  const tile: Tile = {
    id: newId(),
    value: Math.random() < 0.9 ? 2 : 4,
    row: r, col: c,
    isNew: true,
  };
  return { ...state, tiles: [...state.tiles, tile] };
}

export function newGame(size = 4): GameState {
  nextId = 1;
  let s: GameState = {
    tiles: [],
    size,
    score: 0,
    best: 0,
    status: 'playing',
    hasReached2048: false,
    history: [],
  };
  s = spawnTile(s);
  s = spawnTile(s);
  return s;
}

/** Traverse order so tiles closest to the moving edge are processed first. */
function traversals(size: number, dir: Direction) {
  const rows = [...Array(size).keys()];
  const cols = [...Array(size).keys()];
  if (dir === 'down')  rows.reverse();
  if (dir === 'right') cols.reverse();
  return { rows, cols };
}

function vector(dir: Direction): [number, number] {
  switch (dir) {
    case 'up':    return [-1, 0];
    case 'down':  return [ 1, 0];
    case 'left':  return [ 0,-1];
    case 'right': return [ 0, 1];
  }
}

function findFarthest(
  grid: (Tile | null)[][],
  r: number, c: number,
  dr: number, dc: number,
) {
  let pr = r, pc = c;
  while (true) {
    const nr = pr + dr, nc = pc + dc;
    if (nr < 0 || nc < 0 || nr >= grid.length || nc >= grid.length) break;
    if (grid[nr][nc]) return { farthest: { r: pr, c: pc }, next: { r: nr, c: nc } };
    pr = nr; pc = nc;
  }
  return { farthest: { r: pr, c: pc }, next: null as null | { r: number; c: number } };
}

/**
 * Apply a move. Returns:
 *   moved      — true if anything actually changed (so we should spawn a new tile)
 *   tiles      — the new canonical tile list, each carrying prevRow/prevCol for animation
 *   gainedScore
 */
export function move(state: GameState, dir: Direction): {
  state: GameState;
  moved: boolean;
  gainedScore: number;
} {
  const { size } = state;
  // 1) Build working grid; clear per-move flags; remember prev positions.
  const tiles = state.tiles.map(t => ({
    ...t,
    prevRow: t.row,
    prevCol: t.col,
    isNew: false,
    mergedFromIds: undefined,
  }));
  const grid = gridFromTiles(size, tiles);

  let moved = false;
  let gained = 0;
  const [dr, dc] = vector(dir);
  const { rows, cols } = traversals(size, dir);

  // Track which tiles got merged INTO this turn so a result can't merge again.
  const mergedThisTurn = new Set<number>();
  // Tiles that were consumed (their visual representation should fly to the result then disappear).
  const consumed: Tile[] = [];

  for (const r of rows) {
    for (const c of cols) {
      const tile = grid[r][c];
      if (!tile) continue;
      const { farthest, next } = findFarthest(grid, r, c, dr, dc);
      const target = next ? grid[next.r][next.c] : null;

      if (target && target.value === tile.value && !mergedThisTurn.has(target.id)) {
        // Merge — `tile` slides into `target`'s cell; result has a new id (for the pop).
        const resultId = newId();
        const result: Tile = {
          id: resultId,
          value: tile.value * 2,
          row: next!.r, col: next!.c,
          prevRow: target.row, prevCol: target.col, // anchor pop on the target
          mergedFromIds: [tile.id, target.id],
        };
        // Source tile: animate from its old spot to result spot, then disappear.
        const movingSource: Tile = {
          ...tile, row: next!.r, col: next!.c, prevRow: r, prevCol: c,
        };
        consumed.push(movingSource, { ...target });

        // Replace in grid
        grid[r][c] = null;
        grid[next!.r][next!.c] = result;
        mergedThisTurn.add(resultId);

        gained += result.value;
        moved = true;
      } else if (farthest.r !== r || farthest.c !== c) {
        // Slide
        grid[r][c] = null;
        grid[farthest.r][farthest.c] = { ...tile, row: farthest.r, col: farthest.c };
        moved = true;
      }
    }
  }

  if (!moved) {
    return { state, moved: false, gainedScore: 0 };
  }

  // Canonical list = survivors on the grid (+ consumed for one frame of animation).
  const survivors: Tile[] = [];
  for (let r = 0; r < size; r++)
    for (let c = 0; c < size; c++)
      if (grid[r][c]) survivors.push(grid[r][c]!);

  // Order matters for React keys, but we use stable ids so any order is fine.
  const renderTiles = [...consumed, ...survivors];

  const reached = survivors.some(t => t.value >= WIN_VALUE);
  const newScore = state.score + gained;
  const newBest = Math.max(state.best, newScore);
  const wonNow = reached && !state.hasReached2048;

  // Push history for undo (capped) BEFORE applying changes.
  const histSnap = {
    tiles: state.tiles.map(t => ({ ...t })),
    size: state.size,
    score: state.score,
    status: state.status,
    hasReached2048: state.hasReached2048,
  };
  const history = [...state.history, histSnap].slice(-HISTORY_CAP);

  let nextState: GameState = {
    ...state,
    tiles: renderTiles,
    score: newScore,
    best: newBest,
    hasReached2048: state.hasReached2048 || reached,
    status: wonNow ? 'won' : state.status,
    history,
  };

  return { state: nextState, moved: true, gainedScore: gained };
}

/** Drop consumed (mergedFromIds) tiles after the animation completes. */
export function settle(state: GameState): GameState {
  const tiles = state.tiles.filter(t => !t.mergedFromIds || isResult(state, t));
  // Result tiles have mergedFromIds AND occupy a real grid cell — we keep both shape-wise,
  // but in our model `consumed` sources are the ones we drop. We can identify them by
  // not appearing on the grid built from the highest id per cell.
  const grid = gridFromTiles(state.size, [...state.tiles].sort((a,b) => a.id - b.id));
  const survivorIds = new Set<number>();
  for (let r = 0; r < state.size; r++)
    for (let c = 0; c < state.size; c++)
      if (grid[r][c]) survivorIds.add(grid[r][c]!.id);
  return { ...state, tiles: state.tiles.filter(t => survivorIds.has(t.id)) };
}
function isResult(state: GameState, t: Tile) {
  return state.tiles.some(x => x.row === t.row && x.col === t.col && x.id === t.id);
}

export function canMove(state: GameState): boolean {
  const g = gridFromTiles(state.size, state.tiles.filter(t => !t.mergedFromIds));
  for (let r = 0; r < state.size; r++) {
    for (let c = 0; c < state.size; c++) {
      const t = g[r][c];
      if (!t) return true;
      for (const [dr, dc] of [[1,0],[0,1]] as const) {
        const nr = r + dr, nc = c + dc;
        if (nr < state.size && nc < state.size) {
          const n = g[nr][nc];
          if (!n || n.value === t.value) return true;
        }
      }
    }
  }
  return false;
}

export function undo(state: GameState): GameState {
  const last = state.history[state.history.length - 1];
  if (!last) return state;
  return {
    ...state,
    ...last,
    best: state.best,
    history: state.history.slice(0, -1),
    status: 'playing',
  };
}

export function continuePlay(state: GameState): GameState {
  return { ...state, status: 'continue' };
}
```

> **Note on `settle`:** the move function returns *both* the consumed source
> tiles and their merge result so the UI can animate them simultaneously
> (sources slide into the result cell, result pops). Once the animation
> finishes, call `settle()` to drop the consumed tiles. `useGame` handles
> this automatically below.

## 6. `useGame` Hook

**`src/game/useGame.ts`**

```ts
import { useCallback, useEffect, useRef, useState } from 'react';
import * as Haptics from 'expo-haptics';
import { Direction, GameState } from './types';
import { canMove, continuePlay, move, newGame, settle, spawnTile, undo } from './engine';
import { loadSave, saveGame, loadBest, saveBest } from '../storage/save';

const ANIM_MS = 130;    // tile slide duration
const SPAWN_DELAY = ANIM_MS + 20;

export function useGame(size = 4) {
  const [state, setState] = useState<GameState>(() => newGame(size));
  const spawnTimer = useRef<ReturnType<typeof setTimeout> | null>(null);

  // Hydrate from disk on mount
  useEffect(() => {
    (async () => {
      const saved = await loadSave();
      const best = (await loadBest()) ?? 0;
      if (saved) setState({ ...saved, best });
      else setState(s => ({ ...s, best }));
    })();
    return () => { if (spawnTimer.current) clearTimeout(spawnTimer.current); };
  }, []);

  // Persist
  useEffect(() => { saveGame(state); }, [state]);
  useEffect(() => { saveBest(state.best); }, [state.best]);

  const swipe = useCallback((dir: Direction) => {
    setState(prev => {
      if (prev.status === 'lost' || prev.status === 'won') return prev;
      const { state: moved, moved: didMove, gainedScore } = move(prev, dir);
      if (!didMove) return prev;
      if (gainedScore > 0) {
        Haptics.impactAsync(Haptics.ImpactFeedbackStyle.Light).catch(() => {});
      }
      // After animation completes, drop consumed tiles and spawn a new one.
      if (spawnTimer.current) clearTimeout(spawnTimer.current);
      spawnTimer.current = setTimeout(() => {
        setState(s => {
          const settled = settle(s);
          const spawned = spawnTile(settled);
          const lost = !canMove(spawned) ? { status: 'lost' as const } : {};
          return { ...spawned, ...lost };
        });
      }, SPAWN_DELAY);
      return moved;
    });
  }, []);

  const restart = useCallback(() => {
    if (spawnTimer.current) clearTimeout(spawnTimer.current);
    setState(s => ({ ...newGame(size), best: s.best }));
  }, [size]);

  const undoMove = useCallback(() => setState(s => undo(s)), []);
  const keepGoing = useCallback(() => setState(s => continuePlay(s)), []);

  return { state, swipe, restart, undo: undoMove, keepGoing, animMs: ANIM_MS };
}
```

## 7. UI Components

### `src/components/Tile.tsx` — animated tile (slide + pop)

```tsx
import React, { useEffect } from 'react';
import { Text, View } from 'react-native';
import Animated, {
  useSharedValue, useAnimatedStyle, withTiming, withSequence, Easing,
} from 'react-native-reanimated';
import { TILE_COLORS, tileFontSize } from '../theme';
import { Tile as TileT } from '../game/types';

type Props = {
  tile: TileT;
  tileSize: number;
  gap: number;
  animMs: number;
};

const posFor = (idx: number | undefined, tileSize: number, gap: number) =>
  (idx ?? 0) * (tileSize + gap) + gap;

export const Tile: React.FC<Props> = ({ tile, tileSize, gap, animMs }) => {
  const x = useSharedValue(posFor(tile.prevCol ?? tile.col, tileSize, gap));
  const y = useSharedValue(posFor(tile.prevRow ?? tile.row, tileSize, gap));
  const scale = useSharedValue(tile.isNew ? 0 : 1);

  // Slide to current position whenever row/col changes
  useEffect(() => {
    x.value = withTiming(posFor(tile.col, tileSize, gap), { duration: animMs, easing: Easing.out(Easing.cubic) });
    y.value = withTiming(posFor(tile.row, tileSize, gap), { duration: animMs, easing: Easing.out(Easing.cubic) });
  }, [tile.row, tile.col, tileSize, gap, animMs]);

  // Pop animation
  useEffect(() => {
    if (tile.isNew) {
      scale.value = withTiming(1, { duration: 140, easing: Easing.out(Easing.back(1.6)) });
    } else if (tile.mergedFromIds) {
      scale.value = withSequence(
        withTiming(1.18, { duration: animMs * 0.6, easing: Easing.out(Easing.cubic) }),
        withTiming(1,    { duration: animMs * 0.6, easing: Easing.in(Easing.cubic) }),
      );
    }
  }, [tile.isNew, tile.mergedFromIds, animMs]);

  const aStyle = useAnimatedStyle(() => ({
    transform: [{ translateX: x.value }, { translateY: y.value }, { scale: scale.value }],
  }));

  const palette = TILE_COLORS[tile.value] ?? TILE_COLORS[8192];

  return (
    <Animated.View
      pointerEvents="none"
      style={[
        {
          position: 'absolute',
          width: tileSize, height: tileSize,
          borderRadius: 6,
          backgroundColor: palette.bg,
          alignItems: 'center', justifyContent: 'center',
        },
        aStyle,
      ]}
    >
      <Text style={{
        color: palette.fg,
        fontSize: tileFontSize(tile.value, tileSize),
        fontWeight: '800',
      }}>
        {tile.value}
      </Text>
    </Animated.View>
  );
};
```

### `src/components/Board.tsx` — grid background + swipe handler

```tsx
import React from 'react';
import { View, useWindowDimensions } from 'react-native';
import { Gesture, GestureDetector } from 'react-native-gesture-handler';
import { runOnJS } from 'react-native-reanimated';
import { COLORS } from '../theme';
import { Tile } from './Tile';
import { Direction, GameState } from '../game/types';

const SWIPE_THRESHOLD = 18; // px before we commit to a direction

type Props = {
  state: GameState;
  onSwipe: (dir: Direction) => void;
  animMs: number;
  maxWidth?: number;
};

export const Board: React.FC<Props> = ({ state, onSwipe, animMs, maxWidth }) => {
  const { width } = useWindowDimensions();
  const boardSize = Math.min(maxWidth ?? Infinity, width - 24);
  const gap = Math.round(boardSize * 0.025);
  const tileSize = Math.floor((boardSize - gap * (state.size + 1)) / state.size);
  const inner = tileSize * state.size + gap * (state.size + 1);

  const pan = Gesture.Pan()
    .minDistance(SWIPE_THRESHOLD)
    .onEnd(e => {
      'worklet';
      const { translationX: dx, translationY: dy } = e;
      const ax = Math.abs(dx), ay = Math.abs(dy);
      let dir: Direction;
      if (ax > ay) dir = dx > 0 ? 'right' : 'left';
      else         dir = dy > 0 ? 'down'  : 'up';
      runOnJS(onSwipe)(dir);
    });

  // Empty-cell background grid
  const cells = [];
  for (let r = 0; r < state.size; r++) {
    for (let c = 0; c < state.size; c++) {
      cells.push(
        <View
          key={`cell-${r}-${c}`}
          style={{
            position: 'absolute',
            top: r * (tileSize + gap) + gap,
            left: c * (tileSize + gap) + gap,
            width: tileSize, height: tileSize,
            borderRadius: 6,
            backgroundColor: COLORS.cellEmpty,
          }}
        />
      );
    }
  }

  return (
    <GestureDetector gesture={pan}>
      <View
        style={{
          width: inner, height: inner,
          backgroundColor: COLORS.boardBg,
          borderRadius: 8,
          alignSelf: 'center',
        }}
      >
        {cells}
        {state.tiles.map(t => (
          <Tile key={t.id} tile={t} tileSize={tileSize} gap={gap} animMs={animMs} />
        ))}
      </View>
    </GestureDetector>
  );
};
```

### `src/components/ScoreBox.tsx`

```tsx
import React, { useEffect, useRef, useState } from 'react';
import { Text, View } from 'react-native';
import Animated, {
  useSharedValue, useAnimatedStyle, withSequence, withTiming, Easing, runOnJS,
} from 'react-native-reanimated';
import { COLORS } from '../theme';

export const ScoreBox: React.FC<{ label: string; value: number }> = ({ label, value }) => {
  const prev = useRef(value);
  const [delta, setDelta] = useState<number | null>(null);
  const opacity = useSharedValue(0);
  const ty = useSharedValue(0);

  useEffect(() => {
    const diff = value - prev.current;
    prev.current = value;
    if (diff > 0) {
      setDelta(diff);
      opacity.value = 1; ty.value = 0;
      opacity.value = withSequence(withTiming(1, { duration: 50 }), withTiming(0, { duration: 500 }));
      ty.value = withTiming(-24, { duration: 550, easing: Easing.out(Easing.cubic) }, () => {
        runOnJS(setDelta)(null);
      });
    }
  }, [value]);

  const aStyle = useAnimatedStyle(() => ({
    opacity: opacity.value, transform: [{ translateY: ty.value }],
  }));

  return (
    <View style={{
      backgroundColor: COLORS.scoreBg,
      paddingHorizontal: 14, paddingVertical: 6,
      borderRadius: 4, minWidth: 80, alignItems: 'center',
    }}>
      <Text style={{ color: COLORS.scoreText, fontSize: 10, fontWeight: '700', letterSpacing: 1 }}>
        {label}
      </Text>
      <Text style={{ color: '#fff', fontSize: 22, fontWeight: '800' }}>{value}</Text>
      {delta !== null && (
        <Animated.Text style={[
          { position: 'absolute', top: 6, right: 6, color: '#776e65', fontWeight: '800', fontSize: 16 },
          aStyle,
        ]}>
          +{delta}
        </Animated.Text>
      )}
    </View>
  );
};
```

## 8. Persistence

**`src/storage/save.ts`**

```ts
import AsyncStorage from '@react-native-async-storage/async-storage';
import { GameState } from '../game/types';

const SAVE_KEY = '2048:save';
const BEST_KEY = '2048:best';

export async function saveGame(s: GameState) {
  const slim = {
    tiles: s.tiles.filter(t => !t.mergedFromIds).map(t => ({
      id: t.id, value: t.value, row: t.row, col: t.col,
    })),
    size: s.size,
    score: s.score,
    status: s.status,
    hasReached2048: s.hasReached2048,
    history: [], // don't persist undo across sessions
  };
  await AsyncStorage.setItem(SAVE_KEY, JSON.stringify(slim));
}

export async function loadSave(): Promise<GameState | null> {
  const raw = await AsyncStorage.getItem(SAVE_KEY);
  if (!raw) return null;
  try {
    const slim = JSON.parse(raw);
    return { ...slim, best: 0, history: [] } as GameState;
  } catch { return null; }
}

export async function loadBest(): Promise<number | null> {
  const raw = await AsyncStorage.getItem(BEST_KEY);
  return raw ? Number(raw) : null;
}

export async function saveBest(v: number) {
  await AsyncStorage.setItem(BEST_KEY, String(v));
}
```

## 12. Options & Variants

| Option | Where it lives | How to enable |
|--------|----------------|---------------|
| **Board size** (3×3, 5×5, 6×6) | `useGame(size)` | Pass `5` or `6` to `useGame` and to a settings UI |
| **Animation speed** | `ANIM_MS` in `useGame.ts` | 90 ms = snappy · 180 ms = chill |
| **Undo depth** | `HISTORY_CAP` in `engine.ts` | Raise to 5–10 for "free undo" mode |
| **4-spawn frequency** | `spawnTile()` — `< 0.9 ? 2 : 4` | Lower the threshold for a harder game |
| **Win threshold** | `WIN_VALUE` in `engine.ts` | Change to 1024 (easy) or 4096 (challenge) |
| **Themes (light/dark)** | `COLORS` + `TILE_COLORS` | Wrap in a context; mirror minesweeper's `themeFor()` pattern |
| **Sound FX** | `expo-av` | Identical pattern to `minesweeper.md` §12 (`sfx.ts`) |
| **Daily seed** | Swap `Math.random` in `spawnTile` with `mulberry32(hash("YYYY-MM-DD"))` | Same RNG module as minesweeper §11 |
